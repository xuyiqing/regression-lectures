\section*{Worksheet}


\begin{enumerate}\itemsep8em
\item (\emph{OLS as sample mean})  
Show that the sample mean $\bar{Y}$ is the OLS estimator of $\mu$ in $Y_i=\mu+u_i$.

\item (\emph{OLS slope as covariance-variance ratio})  
Prove that $\displaystyle \hat{\beta}_1=\frac{\widehat{\mathrm{Cov}}(X,Y)}{\widehat{\mathrm{Var}}(X)}$.

\item (\emph{Geometric orthogonality})  
Show that the OLS residual vector $\hat{u}$ is orthogonal to the regressor vector $x$.

\item (\emph{ANOVA decomposition})  
Show that $SST=SSE+SSR$, where $SST=\sum(y_i-\bar y)^2$, $SSE=\sum(\hat y_i-\bar y)^2$, and $SSR=\sum(y_i-\hat y_i)^2$.

\end{enumerate}
